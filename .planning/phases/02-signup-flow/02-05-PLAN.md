const handleSubmit = async (e: React.FormEvent) => {
  e.preventDefault();
  if (!email || isSubmitting || turnstileBlocked) return;
  try {
    await turnstileRef.current?.execute();
    const token = turnstileRef.current?.getResponse();
    await onSubmit(email, token ?? '');  // empty token → server-side turnstile_failed

};

import {
  FunctionsHttpError,
  FunctionsRelayError,
  FunctionsFetchError,
} from '@supabase/supabase-js';

const { data, error } = await supabase.functions.invoke<JoinWaitlistResponse>(
  'join-waitlist',

);
if (error) {
  if (error instanceof FunctionsHttpError) {
    try { return (await error.context.json() as JoinWaitlistResponse).state; }
    catch { return 'server_error'; }

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser (visitor) ↔ `joinWaitlist` client helper | Untrusted user input (`email`) crosses here. HTML5 validation is best-effort; the Edge Function re-validates server-side. |
| Client ↔ Cloudflare Turnstile widget | The widget runs Cloudflare-hosted JS in an iframe. It mints a single-use token that the client passes upstream — the client cannot see Cloudflare's secret. |
| Client ↔ `join-waitlist` Edge Function | The browser sends `{ email, turnstileToken }` over HTTPS with the anon JWT. CORS allows `*`. The function is the authoritative gate. |
| Client bundle ↔ git / Vercel | `VITE_SUPABASE_URL`, `VITE_SUPABASE_ANON_KEY`, `VITE_TURNSTILE_SITE_KEY` are inlined into the bundle. All three are public-safe; the anon key has no service-role permissions, the site key is meant to be public. |
| Module-load env validation ↔ dev server boot | If any of the three `VITE_*` is missing, the supabase.ts module throws at import time — Vite surfaces this immediately rather than a silent createClient("undefined","undefined"). |
